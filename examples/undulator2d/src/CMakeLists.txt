# =============================================================================
# BELFEM user plugins for the undulator2d example
# =============================================================================
# Builds the shared library this deck loads:
#
#   usersource.so  -- the current sources ( current.cpp )
#
# Build:
#   mkdir build && cd build
#   cmake -DBELFEM_DIR=/path/to/belfem .. && make
#
# The deck refers to it as src/build/usersource.so.
# =============================================================================

cmake_minimum_required(VERSION 3.13)

project(Undulator2dUserPlugins CXX)

# -----------------------------------------------------------------------------
# USER CONFIGURATION
# -----------------------------------------------------------------------------

# BELFEM source tree or installation. Override on the command line:
#   cmake -DBELFEM_DIR=/home/you/codes/belfem ..
set(BELFEM_DIR "$ENV{HOME}/codes/belfem" CACHE PATH "BELFEM source or install root")

# Where the plugins read their measured tables from. Compiled in, so the
# libraries work regardless of the directory the solver is launched from.
set(USER_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../data" CACHE PATH "Measured data tables")

# -----------------------------------------------------------------------------
# BELFEM headers
# -----------------------------------------------------------------------------

# Headers install as <prefix>/include/belfem/<module>/..., preserving the module
# layout of the source tree, so BOTH branches need the module list. Probe for a
# header, not for a directory name: "${BELFEM_DIR}/include" exists after any
# `make install`, so a directory probe accepts the tree and the build then dies
# on a missing header instead of here.
set(BELFEM_MODULE_DIRS
    core containers math/graph io
    numerics/sources physics/materials fem/kernel)

if(EXISTS "${BELFEM_DIR}/src/fem/kernel/belfem_user_api.hpp")
    set(BELFEM_HEADER_ROOT "${BELFEM_DIR}/src")
    message(STATUS "Using BELFEM source tree: ${BELFEM_DIR}")
elseif(EXISTS "${BELFEM_DIR}/include/belfem/fem/kernel/belfem_user_api.hpp")
    set(BELFEM_HEADER_ROOT "${BELFEM_DIR}/include/belfem")
    message(STATUS "Using installed BELFEM: ${BELFEM_DIR}")
else()
    message(FATAL_ERROR
        "Cannot find belfem_user_api.hpp under BELFEM_DIR. Expected either\n"
        "  ${BELFEM_DIR}/src/fem/kernel/belfem_user_api.hpp   (source tree), or\n"
        "  ${BELFEM_DIR}/include/belfem/fem/kernel/belfem_user_api.hpp   (installed)\n"
        "Current BELFEM_DIR: ${BELFEM_DIR}")
endif()

set(BELFEM_INCLUDE_DIRS "${BELFEM_HEADER_ROOT}")
foreach(module IN LISTS BELFEM_MODULE_DIRS)
    list(APPEND BELFEM_INCLUDE_DIRS "${BELFEM_HEADER_ROOT}/${module}")
endforeach()

# -----------------------------------------------------------------------------
# The <belfem_user_api> umbrella
# -----------------------------------------------------------------------------
# The sources include the extension-less SDK umbrella, which is what an
# INSTALLED BELFEM is meant to expose. Neither a source tree nor today's
# install ships that spelling -- both carry belfem_user_api.hpp under
# fem/kernel -- so generate a one-line forwarder and put it on the include
# path. The block no-ops once a prefix ships the real header
# (see todo/user_api_header_install.md).
if(NOT EXISTS "${BELFEM_HEADER_ROOT}/belfem_user_api")
    set(BELFEM_USER_API_SHIM "${CMAKE_CURRENT_BINARY_DIR}/api_shim")
    file(MAKE_DIRECTORY "${BELFEM_USER_API_SHIM}")
    file(WRITE "${BELFEM_USER_API_SHIM}/belfem_user_api"
         "// generated by CMakeLists.txt -- forwards to the fem/kernel umbrella\n"
         "#include \"belfem_user_api.hpp\"\n")
    list(APPEND BELFEM_INCLUDE_DIRS "${BELFEM_USER_API_SHIM}")
    message(STATUS "Generated <belfem_user_api> forwarder")
endif()

# -----------------------------------------------------------------------------
# Compiler settings
# -----------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# -----------------------------------------------------------------------------
# The library
# -----------------------------------------------------------------------------
# No "lib" prefix: the deck names it exactly as it appears on disk.

add_library(usersource MODULE current.cpp)

foreach(TARGET usersource)
    target_include_directories(${TARGET} PRIVATE ${BELFEM_INCLUDE_DIRS})
    target_compile_definitions(${TARGET} PRIVATE
        BELFEM_USER_DATA_DIR="${USER_DATA_DIR}/")
    # ".so" on macOS too, NOT ".dylib": the deck's `file :` names it that way,
    # and a MODULE is dlopen()ed by name, never linked against.
    set_target_properties(${TARGET} PROPERTIES PREFIX "" SUFFIX ".so")
    # Nothing is linked here: the host resolves the plugin's undefined BELFEM
    # symbols at dlopen ( Linux: the executables carry -rdynamic ). Mach-O's
    # two-level namespace rejects them at LINK time instead, so it needs the
    # same policy libbelfem itself uses for gComm/gLog
    # ( config/scripts/Add_BelfemLibrary.cmake ).
    if(APPLE)
        target_link_options(${TARGET} PRIVATE "-Wl,-undefined,dynamic_lookup")
    endif()
endforeach()

install(TARGETS usersource LIBRARY DESTINATION lib)

message(STATUS "==================================================")
message(STATUS "BELFEM  : ${BELFEM_DIR}")
message(STATUS "Data    : ${USER_DATA_DIR}")
message(STATUS "Targets : usersource")
message(STATUS "==================================================")
