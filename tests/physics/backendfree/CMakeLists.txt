# Backend-free compile gate for the user-material API (see the header comment
# in test_MaterialBackendFree.cpp for the contract it pins).
#
# The gate lives in its own subdirectory because the strip below is
# directory-scoped: doing it in tests/physics/ itself would also strip
# test_physics, whose YBCO/Jc tests need the backend.

# the backend macro arrives via add_definitions() from BELFEM_DEFS
# (config/compiler/finalize_compiler.cmake); remove_definitions() is its
# directory-scoped inverse. The ARMA_ALLOW_FAKE_* flags only ever accompany
# BELFEM_ARMADILLO (Intel builds) and go with it. All other defines
# (solvers, MPI, DEBUG, ...) stay — the contract is only about the backend.
remove_definitions( -DBELFEM_ARMADILLO -DBELFEM_BLAZE
                    -DARMA_ALLOW_FAKE_GCC -DARMA_ALLOW_FAKE_CLANG )

# explicit include paths: everything the cl_Material.hpp closure needs. Do not
# rely on inherited include directories here — the paths tests currently
# inherit leak from the top-level executable setup, not from Add_Test.cmake.
include_directories( ${BELFEM_SOURCE_DIR}/core )
include_directories( ${BELFEM_SOURCE_DIR}/containers )
include_directories( ${BELFEM_SOURCE_DIR}/physics/materials )
# the shipped example includes <belfem_user_api>, whose closure reaches these
include_directories( ${BELFEM_SOURCE_DIR}/numerics/sources )
include_directories( ${BELFEM_SOURCE_DIR}/fem/kernel )
# extension-less forwarder to the in-tree umbrella, same shim the plugin
# templates generate; drop it once the install ships the real header
file( MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/api_shim )
file( WRITE ${CMAKE_CURRENT_BINARY_DIR}/api_shim/belfem_user_api
      "// generated by CMakeLists.txt -- forwards to the in-tree umbrella\n"
      "#include \"belfem_user_api.hpp\"\n" )
include_directories( ${CMAKE_CURRENT_BINARY_DIR}/api_shim )

# OBJECT library: compiled on every full build (it is part of ALL), never
# linked, never run — no backend symbols are ever needed. The shipped plugin
# example is compiled here too, so it cannot silently rot.
add_library( test_material_backendfree OBJECT
        test_MaterialBackendFree.cpp
        ${BELFEM_SOURCE_DIR}/physics/materials/example_user_material.cpp
        )

# make check builds test_physics; pull the gate along with it
add_dependencies( test_physics test_material_backendfree )
